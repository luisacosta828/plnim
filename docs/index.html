cosine_similarity.sql 
-- AI & Vector Embeddings: Compute Cosine Similarity at bare-metal C speed
CREATE FUNCTION cosine_similarity(v1 float8[], v2 float8[]) RETURNS float8 AS $$
  import std/math
  var dotProduct = 0.0
  var normA = 0.0
  var normB = 0.0
  for i in 0 ..< min(v1.len, v2.len):
    dotProduct += v1[i] * v2[i]
    normA += v1[i] * v1[i]
    normB += v2[i] * v2[i]
  let denominator = sqrt(normA) * sqrt(normB)
  return if denominator == 0.0: 0.0 else: dotProduct / denominator
$$ LANGUAGE plnim;

SELECT cosine_similarity(ARRAY[0.1, 0.8, 0.3], ARRAY[0.2, 0.7, 0.4]);
-- Result: 0.957297 (Microsecond vector calculation directly in Postgres)
levenshtein_distance.sql 
-- High-speed fuzzy string distance across database rows
CREATE FUNCTION levenshtein_dist(s1 text, s2 text) RETURNS int AS $$
  import std/strutils
  return editDistance(s1, s2).int32
$$ LANGUAGE plnim;

SELECT levenshtein_dist('PostgreSQL', 'Postgres');
-- Result: 2 (100x faster than interpreted PL/pgSQL loops)

-- Compute 64-bit token hash using Nim's std/hashes library
CREATE FUNCTION fast_hash(key text) RETURNS int64 AS $$
  import std/hashes

point_distance.sql 
CREATE TYPE geo_point AS (
  x float8,
  y float8
);

-- Zero boilerplate: geo_point automatically maps to Nim's Geo_point struct
CREATE FUNCTION point_distance(p1 geo_point, p2 geo_point) RETURNS float8 AS $$
  import std/math
  let dx = p1.x - p2.x
  let dy = p1.y - p2.y
  return sqrt(dx * dx + dy * dy)
$$ LANGUAGE plnim;

SELECT point_distance(ROW(1.0, 2.0)::geo_point, ROW(4.0, 6.0)::geo_point);
-- Result: 5.0
jsonb_telemetry.sql 
-- First-class integration with Nim's std/json and JsonNode constructors
CREATE FUNCTION generate_telemetry(server text, cpu float8, mem float8) RETURNS jsonb AS $$
  return %*{
    "server": server,
    "metrics": {
      "cpu_pct": cpu,
      "mem_pct": mem
    },
    "healthy": cpu < 85.0
  }
$$ LANGUAGE plnim;

SELECT generate_telemetry('db-master-01', 45.5, 60.2);
-- Result: {"healthy": true, "metrics": {"cpu_pct": 45.5, "mem_pct": 60.2}, "server": "db-master-01"}
arrays_seq.sql 
-- Native PostgreSQL int[] maps directly to Nim's seq[int32]
CREATE FUNCTION sum_scores(scores int[]) RETURNS int AS $$
  import std/sequtils
  return foldl(scores, a + b, 0)
$$ LANGUAGE plnim;

SELECT sum_scores(ARRAY[10, 20, 30, 40, 50]);
-- Result: 150
panic_shield.sql 
-- In C extensions, an unhandled exception causes SIGSEGV and crashes PostgreSQL.
-- PL/Nim's Panic Shield intercepts all Defect exceptions into PostgreSQL ERRORs!
CREATE FUNCTION safe_index_lookup(items text[], idx int) RETURNS text AS $$
  return items[idx]
$$ LANGUAGE plnim;

-- Safely reports: ERROR: Extension Defect [IndexDefect]: index out of bounds
-- PostgreSQL worker process remains completely healthy!

Native C Speed

Compiles directly to optimized C shared objects via GCC/Clang with ORC memory management. Zero interpreter overhead.

🛡️

Crash-Proof Panic Shield

Catches all memory defects, overflows, and out-of-bounds access, safely converting them into PostgreSQL transaction rollbacks.

Natural Import Hoisting

Write standard Nim modules (std/math, std/json, std/strutils) inside functions without worrying about module scope.

Quick Start with Docker

Run PostgreSQL with PL/Nim in a single command.

bash terminal

$ git clone https://github.com/luisacosta828/plnim.git

$ cd plnim

$ docker-compose -f docker/docker-compose.yml up -d

# PostgreSQL is now running with PL/Nim enabled on port 5445!

$ psql -h localhost -p 5445 -U postgres
